#!/usr/bin/env bash
set -u

URL='http://127.0.0.1:1984/api/frame.jpeg?src=camera_giardino_anteriore'
OUT='/run/bottazzi-garden-warmer-frame.jpg'

while true; do
  if timeout 12 curl -fsS "$URL" -o "$OUT.tmp"; then
    mv "$OUT.tmp" "$OUT"
  else
    rm -f "$OUT.tmp"
    echo "frame_warmer_curl_failed $(date --iso-8601=seconds)" >&2
  fi
  sleep 1
done
